local function scanRemotes()
    if _scanDone then return end
    _scanDone=true
    local rs=game:GetService("ReplicatedStorage")
    local hdNames={"HDAdminRemote","MainRemote","HDAdmin","HDAdminEvent","AdminRemote","HD_Admin","HDAdminRE","Commands"}
    local f3xNames={"F3XRemote","BuildRemote","Remote","BuildService"}
    for _,n in ipairs(hdNames)do
        local obj=rs:FindFirstChild(n)
        if obj and obj:IsA("RemoteEvent")then _hdRemote=obj break end
    end
    for _,n in ipairs(f3xNames)do
        local obj=rs:FindFirstChild(n)
        if obj and obj:IsA("RemoteEvent")then _f3xRemote=obj break end
    end
    local notif=Instance.new("Frame",sg)
    notif.Size=UDim2.new(0,300,0,36)
    notif.Position=UDim2.new(0.5,-150,1,10)
    notif.BackgroundColor3=HDR
    notif.BorderSizePixel=2
    notif.BorderColor3=BD
    notif.ZIndex=999
    local ntxt=Instance.new("TextLabel",notif)
    ntxt.Size=UDim2.new(1,-8,1,0)
    ntxt.Position=UDim2.new(0,4,0,0)
    ntxt.BackgroundTransparency=1
    ntxt.TextColor3=TXT
    ntxt.Font=Enum.Font.GothamBold
    ntxt.TextSize=10
    ntxt.TextWrapped=true
    ntxt.TextXAlignment=Enum.TextXAlignment.Left
    ntxt.ZIndex=999
    local hds=_hdRemote and("OK:".._hdRemote.Name)or"NAO ACHOU"
    local f3s=_f3xRemote and("OK:".._f3xRemote.Name)or"NAO ACHOU"
    ntxt.Text="[SCAN] HD:"..hds.." | F3X:"..f3s
    T:Create(notif,TweenInfo.new(0.4),{Position=UDim2.new(0.5,-150,1,-50)}):Play()
    task.wait(5)
    T:Create(notif,TweenInfo.new(0.4),{Position=UDim2.new(0.5,-150,1,10)}):Play()
    task.wait(0.5)
    notif:Destroy()
end
task.spawn(scanRemotes)
local function hdCmd(cmd)
    if not _hdRemote then scanRemotes() end
    if _hdRemote then pcall(function()_hdRemote:FireServer(cmd)end) end
end
local function f3xFire(a,b,c,d)
    if not _f3xRemote then scanRemotes() end
    if _f3xRemote then pcall(function()_f3xRemote:FireServer(a,b,c,d)end) end
end
local function f3xProp(obj,prop,val)
    f3xFire("ChangeProp",obj,prop,val)
    pcall(function()
        if prop=="BrickColor" then obj.BrickColor=val
        elseif prop=="Material" then obj.Material=val
        elseif prop=="Anchored" then obj.Anchored=val
        end
    end)
end

local function launchSky(skyId)
    local player=lp
    local char=player.Character or player.CharacterAdded:Wait()
    local tool
    for _,v in ipairs(player:GetDescendants())do
        if v.Name=="SyncAPI" then tool=v.Parent end
    end
    for _,v in ipairs(game:GetService("ReplicatedStorage"):GetDescendants())do
        if v.Name=="SyncAPI" then tool=v.Parent end
    end
    if not tool then return end
    local remote=tool.SyncAPI.ServerEndpoint
    local RS2=game:GetService("RunService")
    local function _(args) remote:InvokeServer(unpack(args)) end
    local function CreatePart(cf,parent) _({[1]="CreatePart",[2]="Normal",[3]=cf,[4]=parent}) end
    local function SetAnchor(part,bool) _({[1]="SyncAnchor",[2]={{["Part"]=part,["Anchored"]=bool}}}) end
    local function AddMesh(part) _({[1]="CreateMeshes",[2]={{["Part"]=part}}}) end
    local function SetMesh(part,id) _({[1]="SyncMesh",[2]={{["Part"]=part,["MeshId"]="rbxassetid://"..id}}}) end
    local function SetTexture(part,tex) _({[1]="SyncMesh",[2]={{["Part"]=part,["TextureId"]=tex}}}) end
    local function MeshResize(part,size) _({[1]="SyncMesh",[2]={{["Part"]=part,["Scale"]=size}}}) end
    local function SetTransparency(part,val) _({[1]="SyncTransparency",[2]={{["Part"]=part,["Transparency"]=val}}}) end
    local images={}
    for i=1,8 do images[i]="http://www.roblox.com/asset/?id="..skyId end
    local skyPart,skyLoop=nil,nil
    local frameTime,lastUpdate=1/10,0
    local function CreateSky()
        local hrp=char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local cf=hrp.CFrame
        CreatePart(CFrame.new(cf.Position+Vector3.new(0,6,0)),workspace)
        local found=false
        for i=1,50 do
            task.wait()
            for _,v in ipairs(workspace:GetDescendants())do
                if v:IsA("BasePart")and(v.Position-(cf.Position+Vector3.new(0,6,0))).Magnitude<1 then
                    skyPart=v found=true break
                end
            end
            if found then break end
        end
        if not skyPart then return end
        SetAnchor(skyPart,true)
        AddMesh(skyPart)
        SetMesh(skyPart,"111891702759441")
        MeshResize(skyPart,Vector3.new(9000,9000,9000))
        SetTransparency(skyPart,0)
        local index=1
        skyLoop=RS2.Heartbeat:Connect(function(dt)
            lastUpdate=lastUpdate+dt
            if lastUpdate>=frameTime then
                lastUpdate=0
                if not skyPart then skyLoop:Disconnect() return end
                SetTexture(skyPart,images[index])
                index=(index%#images)+1
            end
        end)
    end
    task.spawn(CreateSky)
end

local function tweenTo(hrp,targetPos,maxTime)
    maxTime=maxTime or 2.4
    local dist=(hrp.Position-targetPos).Magnitude
    local speed=math.max(dist/maxTime,22)
    local hum=hrp.Parent and hrp.Parent:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed=speed end
    local bp=Instance.new("BodyPosition",hrp)
    bp.MaxForce=Vector3.new(1e6,1e6,1e6)
    bp.D=1000 bp.P=50000 bp.Position=targetPos
    local timeout=0
    repeat
        task.wait(0.05) timeout=timeout+0.05
        if(hrp.Position-targetPos).Magnitude<4 then break end
    until timeout>=maxTime
    if bp and bp.Parent then bp:Destroy() end
    if hum then hum.WalkSpeed=16 end
end
local function getBestCollect()
    local plots=workspace:FindFirstChild("Plots")
    if not plots then return nil,nil,nil end
    local best,bestMult,bestPos,bestName=nil,0,nil,""
    for _,plot in ipairs(plots:GetChildren())do
        if not plot:IsA("Model")then continue end
        if _myPlotRef and plot==_myPlotRef then continue end
        local occupied=false
        for _,v in ipairs(plot:GetDescendants())do
            if v:IsA("TextLabel")and string.find(tostring(v.Text),"'s Base")then occupied=true break end
        end
        if not occupied then continue end
        local multVal=0
        for _,v in ipairs(plot:GetDescendants())do
            if v:IsA("TextLabel")then
                local n=tonumber(tostring(v.Text):match("x(%d+)"))
                if n and n>multVal then multVal=n end
            end
        end
        local cash=plot:FindFirstChild("CashPad",true)
        if cash and cash:IsA("BasePart")then
            if multVal>=bestMult then
                bestMult=multVal best=plot
                bestPos=cash.Position+Vector3.new(0,3,0)
            end
        end
    end
    return best,bestPos,bestName
end
